cmake_minimum_required(VERSION 3.16)
project(estun_anygrasp_motion)
set(CMAKE_CXX_STANDARD 17)
set(Python3_EXECUTABLE /usr/bin/python3 CACHE FILEPATH "ROS Python")
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(codroid_msgs REQUIRED)
find_package(step_motor REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
add_executable(anygrasp_pregrasp src/anygrasp_pregrasp.cpp)
ament_target_dependencies(anygrasp_pregrasp rclcpp codroid_msgs sensor_msgs trajectory_msgs moveit_ros_planning_interface moveit_ros_planning moveit_msgs)
target_link_libraries(anygrasp_pregrasp yaml-cpp)
add_executable(anygrasp_positioning src/anygrasp_pregrasp.cpp)
target_compile_definitions(anygrasp_positioning PRIVATE ANYGRASP_POSITIONING_ONLY)
ament_target_dependencies(anygrasp_positioning rclcpp codroid_msgs sensor_msgs trajectory_msgs moveit_ros_planning_interface moveit_ros_planning moveit_msgs)
target_link_libraries(anygrasp_positioning yaml-cpp)
add_executable(anygrasp_cycle src/anygrasp_pregrasp.cpp)
target_compile_definitions(anygrasp_cycle PRIVATE ANYGRASP_FULL_CYCLE)
ament_target_dependencies(anygrasp_cycle rclcpp codroid_msgs step_motor sensor_msgs trajectory_msgs moveit_ros_planning_interface moveit_ros_planning moveit_msgs)
target_link_libraries(anygrasp_cycle yaml-cpp)
install(TARGETS anygrasp_pregrasp anygrasp_positioning anygrasp_cycle DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/offline_state.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 enable_testing()
 find_package(Threads REQUIRED)
 add_executable(test_robot_status test/test_robot_status.cpp)
 target_include_directories(test_robot_status PRIVATE src)
 target_link_libraries(test_robot_status Threads::Threads)
 add_test(NAME robot_status_startup COMMAND test_robot_status)
 add_executable(test_blended test/test_blended.cpp)
 target_include_directories(test_blended PRIVATE src)
 add_test(NAME blended_contract COMMAND test_blended)
endif()
ament_package()
